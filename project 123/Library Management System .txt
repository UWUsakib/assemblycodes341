; =============================================================================
; FEATURE 4: BOOK RETURN WITH FINE CALCULATION
; =============================================================================
FEATURE4_FINE PROC
    PRINT_STR fineHeader
    PRINT_STR finePrompt
    CALL READ_INPUT


    LEA  SI, inputBuf + 2
    CALL TO_UPPER_SI


    LEA  DI, inputBuf + 2
    LEA  SI, book1Title
    CALL CMP_UPPER
    JNE  F4_C2
    CMP  book1Status, 0
    JNE  F4_NOT_BORROWED
    MOV  book1Status, 1
    JMP  F4_CALC_FINE
F4_C2:
    LEA  DI, inputBuf + 2
    LEA  SI, book2Title
    CALL CMP_UPPER
    JNE  F4_C3
    CMP  book2Status, 0
    JNE  F4_NOT_BORROWED
    MOV  book2Status, 1
    JMP  F4_CALC_FINE
F4_C3:
    LEA  DI, inputBuf + 2
    LEA  SI, book3Title
    CALL CMP_UPPER
    JNE  F4_C4
    CMP  book3Status, 0
    JNE  F4_NOT_BORROWED
    MOV  book3Status, 1
    JMP  F4_CALC_FINE
F4_C4:
    LEA  DI, inputBuf + 2
    LEA  SI, book4Title
    CALL CMP_UPPER
    JNE  F4_C5
    CMP  book4Status, 0
    JNE  F4_NOT_BORROWED
    MOV  book4Status, 1
    JMP  F4_CALC_FINE
F4_C5:
    LEA  DI, inputBuf + 2
    LEA  SI, book5Title
    CALL CMP_UPPER
    JNE  F4_NOT_FOUND
    CMP  book5Status, 0
    JNE  F4_NOT_BORROWED
    MOV  book5Status, 1


F4_CALC_FINE:
    PRINT_STR returnOK
    PRINT_STR fineDaysMsg
    CALL READ_DAYS


    MOV  AX, daysLate
     CMP  AX, 0
    JE   F4_NO_FINE
     CMP  AX, 1
    JE   F4_NO_FINE
    CMP  AX, 2
    JE   F4_NO_FINE
    
    
    SUB AX,2
    MOV  BL, finePerDay
    MOV  BH, 0
    MUL  BX
    MOV  fineResult, AX


    PRINT_STR fineMsg
    MOV  AX, fineResult
    CALL PRINT_NUM_WORD
    PRINT_STR fineTaka
    RET


F4_NO_FINE:
    PRINT_STR noFineMsg
    RET
F4_NOT_BORROWED:
    PRINT_STR notBorrowed
    RET
F4_NOT_FOUND:
    PRINT_STR bookNF
    RET
FEATURE4_FINE ENDP


; =============================================================================
; FEATURE 5: BOOK RATING SYSTEM
;
; Logic:
;   - Displays top-rated books (all books with their average ratings)
;   - User enters a book title
;   - Book must be AVAILABLE (status=1) meaning it was already returned
;   - Accepts a single digit 1-5
;   - Adds to RatingSum, increments RatingCount
;   - Calculates and displays integer average (RatingSum / RatingCount)
;   - Shows "Top Rated Books" list with averages at the end
; =============================================================================
FEATURE5_RATING PROC
F5R_DISPLAY:
    PRINT_STR rateHeader


    ; ---- Show current ratings for all books ----
    PRINT_STR rateTopHeader


    ; Book 1
    PRINT_STR labelTitle
    PRINT_STR book1Title
    PRINT_STR CRLF
    CMP  book1RatingCount, 0
    JE   F5R_B1_NONE
    PRINT_STR rateAvgLabel
    MOV  AL, book1RatingSum
    MOV  AH, 0
    MOV  BL, book1RatingCount
    MOV  BH, 0
    DIV  BL                      ; AL = average
    CALL PRINT_NUM_BYTE
    PRINT_STR rateSlash
    PRINT_STR rateFive
    PRINT_STR CRLF
    PRINT_STR rateCountLabel
    MOV  AL, book1RatingCount
    CALL PRINT_NUM_BYTE
    PRINT_STR CRLF
    JMP  F5R_B2
F5R_B1_NONE:
    PRINT_STR rateNoRating
    PRINT_STR CRLF


    ; Book 2
F5R_B2:
    PRINT_STR labelTitle
    PRINT_STR book2Title
    PRINT_STR CRLF
    CMP  book2RatingCount, 0
    JE   F5R_B2_NONE
    PRINT_STR rateAvgLabel
    MOV  AL, book2RatingSum
    MOV  AH, 0
    MOV  BL, book2RatingCount
    MOV  BH, 0
    DIV  BL
    CALL PRINT_NUM_BYTE
    PRINT_STR rateSlash
    PRINT_STR rateFive
    PRINT_STR CRLF
    PRINT_STR rateCountLabel
    MOV  AL, book2RatingCount
    CALL PRINT_NUM_BYTE
    PRINT_STR CRLF
    JMP  F5R_B3
F5R_B2_NONE:
    PRINT_STR rateNoRating
    PRINT_STR CRLF


    ; Book 3
F5R_B3:
    PRINT_STR labelTitle
    PRINT_STR book3Title
    PRINT_STR CRLF

F5R_B4_NONE:

F5R_B5:

    PRINT_STR rateAvgLabel

    MOV  BH, 0
    DIV  BL
    CALL PRINT_NUM_BYTE
    PRINT_STR rateSlash
    PRINT_STR rateFive
    PRINT_STR CRLF
    PRINT_STR rateCountLabel

    CALL PRINT_NUM_BYTE
    PRINT_STR CRLF
    JMP  F5R_INPUT
F5R_B5_NONE:
    PRINT_STR rateNoRating
    PRINT_STR CRLF


    ; ---- Ask user which book to rate ----
F5R_INPUT:
    PRINT_STR rateSepLine
    PRINT_STR ratePrompt
    CALL READ_INPUT


    LEA  SI, inputBuf + 2
    CALL TO_UPPER_SI


    ; ---- Match book title ----
    LEA  DI, inputBuf + 2
    LEA  SI, book1Title

    JNE  F5R_C2
    ; Book must be Available (returned) to be rated
    CMP  book1Status, 1
    JNE  F5R_NOT_AVAIL
    CALL READ_RATING             ; AL = validated rating digit (1-5), or 0 on invalid
    CMP  AL, 0
    JE   F5R_BAD_RATING
    ADD  book1RatingSum, AL
    INC  book1RatingCount
    JMP  F5R_RATE_DONE


F5R_C2:
    LEA  DI, inputBuf + 2
    LEA  SI, book2Title
    CALL CMP_UPPER
    JNE  F5R_C3

    CALL READ_RATING
    CMP  AL, 0
    JE   F5R_BAD_RATING
    ADD  book2RatingSum, AL
    INC  book2RatingCount
    JMP  F5R_RATE_DONE


F5R_C3:
    LEA  DI, inputBuf + 2

    JNE  F5R_C4
    CMP  book3Status, 1
    JNE  F5R_NOT_AVAIL
    CALL READ_RATING
    CMP  AL, 0
    JE   F5R_BAD_RATING

    JMP  F5R_RATE_DONE

    JNE  F5R_NF
    CMP  book5Status, 1
    JNE  F5R_NOT_AVAIL
    CALL READ_RATING
    CMP  AL, 0
    JE   F5R_BAD_RATING
    ADD  book5RatingSum, AL

F5R_RATE_DONE:

    RET

    RET


F5R_BAD_RATING:

FEATURE5_RATING ENDP


; =============================================================================
; READ_RATING
; Reads a single keypress (1-5). Echoes character.
; Returns: AL = numeric value (1-5) if valid, AL = 0 if invalid.
; =============================================================================
READ_RATING PROC
    PRINT_STR rateScaleMsg


    MOV  AH, 01H
    INT  21H                 ; AL = typed character


    MOV  BL, AL              ; SAVE INPUT


    PRINT_STR CRLF


    MOV  AL, BL              ; RESTORE INPUT


    CMP  AL, '1'
    JB   RR_INVALID


    CMP  AL, '5'
    JA   RR_INVALID


    SUB  AL, '0'
    RET


RR_INVALID:
    MOV  AL, 0
    RET
READ_RATING ENDP

; =============================================================================
; READ_DAYS
; =============================================================================
READ_DAYS PROC
    MOV  daysLate, 0
    MOV  tempWord, 0
    MOV  CX, 0


RD_LOOP:
    MOV  AH, 01H
    INT  21H
    CMP  AL, 0DH
    JE   RD_DONE
    CMP  AL, '0'
    JB   RD_LOOP
    CMP  AL, '9'
    JA   RD_LOOP
    CMP  CX, 2
    JAE  RD_LOOP


    SUB  AL, '0'
    MOV  AH, 0
    PUSH AX
    MOV  AX, tempWord
    MOV  BX, 10
    MUL  BX
    POP  BX
    ADD  AX, BX
    MOV  tempWord, AX
    INC  CX
    JMP  RD_LOOP


RD_DONE:
    MOV  AX, tempWord
    MOV  daysLate, AX
    PRINT_STR CRLF
    RET
READ_DAYS ENDP


; =============================================================================
; TO_UPPER_SI
; =============================================================================
TO_UPPER_SI PROC
TU_LOOP:
    MOV  AL, [SI]
    CMP  AL, '$'
    JE   TU_DONE
    CMP  AL, 'a'
    JB   TU_NEXT
    CMP  AL, 'z'
    JA   TU_NEXT
    SUB  AL, 32
    MOV  [SI], AL
TU_NEXT:
    INC  SI
    JMP  TU_LOOP
TU_DONE:
    RET
TO_UPPER_SI ENDP


; =============================================================================
; CMP_UPPER
; =============================================================================
CMP_UPPER PROC
CU_NEXT:
    MOV  AL, [SI]
    CMP  AL, 'a'
    JB   CU_NO_CONV
    CMP  AL, 'z'
    JA   CU_NO_CONV
    SUB  AL, 32
CU_NO_CONV:
    MOV  BL, [DI]
    CMP  AL, '$'
    JE   CU_CHECK_END
    CMP  AL, BL
    JNE  CU_NOT_EQ
    INC  SI
    INC  DI
    JMP  CU_NEXT
CU_CHECK_END:
    CMP  BL, '$'
    JNE  CU_NOT_EQ
    XOR  AX, AX
    CMP  AX, AX
    RET
CU_NOT_EQ:
    MOV  AX, 1
    OR   AX, AX
    RET
CMP_UPPER ENDP


; =============================================================================
; PARTIAL_MATCH
; =============================================================================
PARTIAL_MATCH PROC
    PUSH SI
    PUSH DI
    PUSH BX
    PUSH CX


    MOV  BX, DI
    MOV  CX, 0
PM_LEN:
    MOV  AL, [BX]
    CMP  AL, '$'
    JE   PM_GOT_LEN
    INC  CX
    INC  BX
    JMP  PM_LEN
PM_GOT_LEN:


PM_OUTER:
    MOV  AL, [SI]
    CMP  AL, '$'
    JE   PM_NO_MATCH


    PUSH SI
    PUSH DI
    PUSH CX


    MOV  BX, CX
PM_INNER:
    CMP  BX, 0
    JE   PM_INNER_MATCH
    MOV  AL, [SI]
    CMP  AL, '$'
    JE   PM_INNER_FAIL
    CMP  AL, 'a'
    JB   PM_NO_LC
    CMP  AL, 'z'
    JA   PM_NO_LC
    SUB  AL, 32
PM_NO_LC:
    MOV  AH, [DI]
    CMP  AL, AH
    JNE  PM_INNER_FAIL
    INC  SI
    INC  DI
    DEC  BX
    JMP  PM_INNER


PM_INNER_MATCH:
    POP  CX
    POP  DI
    POP  SI
    POP  CX
    POP  BX
    POP  DI
    POP  SI
    XOR  AX, AX
    RET


PM_INNER_FAIL:
    POP  CX
    POP  DI
    POP  SI
    INC  SI
    JMP  PM_OUTER


PM_NO_MATCH:
    POP  CX
    POP  BX
    POP  DI
    POP  SI
    MOV  AX, 1
    OR   AX, AX
    RET
PARTIAL_MATCH ENDP


; =============================================================================
; PRINT_NUM_BYTE
; =============================================================================
PRINT_NUM_BYTE PROC
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH DX


    MOV  AH, 0
    MOV  BL, 100
    DIV  BL
    MOV  CH, AL
    MOV  AL, AH
    MOV  AH, 0
    MOV  BL, 10
    DIV  BL
    MOV  CL, AL
    MOV  BL, AH


    MOV  AH, 02H
    CMP  CH, 0
    JE   PNB_TENS
    MOV  DL, CH
    ADD  DL, '0'
    INT  21H
PNB_TENS:
    CMP  CL, 0
    JE   PNB_CHECK_TENS_ZERO
    MOV  DL, CL
    ADD  DL, '0'
    INT  21H
    JMP  PNB_ONES
PNB_CHECK_TENS_ZERO:
    CMP  CH, 0
    JE   PNB_ONES
    MOV  DL, '0'
    INT  21H
PNB_ONES:
    MOV  DL, BL
    ADD  DL, '0'
    INT  21H


    POP  DX
    POP  CX
    POP  BX
    POP  AX
    RET
PRINT_NUM_BYTE ENDP


; =============================================================================
; PRINT_NUM_WORD
; =============================================================================
PRINT_NUM_WORD PROC
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH DX


    MOV  BX, 10
    MOV  CX, 0


    CMP  AX, 0
    JNE  PNW_EXTRACT
    MOV  DL, '0'
    MOV  AH, 02H
    INT  21H
    JMP  PNW_DONE


PNW_EXTRACT:
    CMP  AX, 0
    JE   PNW_PRINT
    MOV  DX, 0
    DIV  BX
    PUSH DX
    INC  CX
    JMP  PNW_EXTRACT


PNW_PRINT:
    CMP  CX, 0
    JE   PNW_DONE
    POP  DX
    ADD  DL, '0'
    MOV  AH, 02H
    INT  21H
    DEC  CX
    JMP  PNW_PRINT


PNW_DONE:
    POP  DX
    POP  CX
    POP  BX
    POP  AX
    RET
PRINT_NUM_WORD ENDP
